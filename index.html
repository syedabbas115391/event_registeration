    // ---------- JavaScript part ----------

    var nameInput = document.getElementById("studentName");
    var emailInput = document.getElementById("studentEmail");
    var eventSelect = document.getElementById("eventSelect");
    var messageBox = document.getElementById("message");

    // 1 and 2. Input validation and dynamic message
    function registerStudent() {
      var name = nameInput.value.trim();
      var email = emailInput.value.trim();
      var eventName = eventSelect.value;
      var errors = [];

      nameInput.classList.remove("is-invalid");
      emailInput.classList.remove("is-invalid");
      eventSelect.classList.remove("is-invalid");

      if (name === "") {
        errors.push("Please enter your name.");
        nameInput.classList.add("is-invalid");
      }
      if (email === "") {
        errors.push("Please enter your email.");
        emailInput.classList.add("is-invalid");
      } else if (email.indexOf("@") === -1 || email.indexOf(".") === -1) {
        errors.push("Please enter a valid email address.");
        emailInput.classList.add("is-invalid");
      }
      if (eventName === "") {
        errors.push("Please select an event.");
        eventSelect.classList.add("is-invalid");
      }

      if (errors.length > 0) {
        messageBox.className = "mt-3 alert alert-danger";
        messageBox.textContent = errors.join(" ");
        return;
      }

      messageBox.className = "mt-3 alert alert-success";
      messageBox.textContent = "Thank you, " + name + "! You are registered for " + eventName + ". A confirmation will be sent to " + email + ".";
    }

    document.getElementById("registerBtn").addEventListener("click", registerStudent);

    // 3. Style manipulation: change card and heading colours
    var themes = [
      { bg: "#ffffff", text: "#12263f", heading: "#12263f" },
      { bg: "#12263f", text: "#ffffff", heading: "#e0a526" },
      { bg: "#e6f4f5", text: "#0a5f67", heading: "#0e7c86" }
    ];
    var themeIndex = 0;

    function changeTheme() {
      themeIndex = (themeIndex + 1) % themes.length;
      var card = document.getElementById("regCard");
      var title = document.getElementById("cardTitle");
      card.style.backgroundColor = themes[themeIndex].bg;
      card.style.color = themes[themeIndex].text;
      title.style.color = themes[themeIndex].heading;
    }

    document.getElementById("themeBtn").addEventListener("click", changeTheme);


    // ---------- jQuery part ----------
    $(document).ready(function () {

      // 1. Show / Hide
      $("#toggleInstructions").click(function () {
        $("#instructions").toggle();
      });

      // 2. jQuery effect: slideToggle
      $("#toggleDetails").click(function () {
        $("#details").slideToggle(400);
      });

      // 3. Dynamic content: add an event to the list and update the counter
      $("#addEventBtn").click(function () {
        var chosen = $("#eventSelect").val();

        if (chosen === "") {
          $("#message").hide().attr("class", "mt-3 alert alert-warning").text("Select an event first, then add it to your list.").fadeIn(300);
          return;
        }

        var alreadyAdded = false;
        $("#myEvents li").each(function () {
          if ($(this).text() === chosen) {
            alreadyAdded = true;
          }
        });

        if (alreadyAdded) {
          $("#message").hide().attr("class", "mt-3 alert alert-info").text(chosen + " is already in your list.").fadeIn(300);
          return;
        }

        $("#emptyItem").remove();
        $("#myEvents").append('<li class="list-group-item"></li>');
        $("#myEvents li:last").text(chosen);
        $("#eventCount").text($("#myEvents li").length);
        $("#message").hide().attr("class", "mt-3 alert alert-success").text(chosen + " was added to your list.").fadeIn(300);
      });

      // 4. Event handling and selectors: mouseover / mouseout and focus / blur
      $("#cardTitle").mouseover(function () {
        $(this).css({ "color": "#e0a526", "letter-spacing": "2px" });
      });
      $("#cardTitle").mouseout(function () {
        $(this).css({ "color": themes[themeIndex].heading, "letter-spacing": "0" });
      });

      $("#myEvents").on("mouseover", "li", function () {
        $(this).css("background-color", "#dff1f2");
      });
      $("#myEvents").on("mouseout", "li", function () {
        $(this).css("background-color", "");
      });

      $(".reg-input").focus(function () {
        $(this).addClass("shadow-sm");
      });
      $(".reg-input").blur(function () {
        $(this).removeClass("shadow-sm");
      });

    });
  </script>
